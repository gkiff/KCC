IGCSE 0984 Pseudocode Reference

                
            

            

                
This interpreter translates standard Cambridge 0984 pseudocode into executable logic. Below are supported structures:

                
                
Variables & Constants

                
DECLARE <identifier> : <data type>
CONSTANT <identifier> = <value>
<identifier> <- <value>  // Assignment


                
Input / Output

                
INPUT <identifier>
OUTPUT <value1>, <value2>, ...


                
Selection (IF/ELSE & CASE)

                
IF <condition> THEN
    // statements
ELSE
    // statements
ENDIF

CASE OF <identifier>
    <value1> : <statement>
    OTHERWISE <statement>
ENDCASE


                
Iteration (Loops)

                
FOR <identifier> <- <val1> TO <val2>
    // statements
NEXT <identifier>

REPEAT
    // statements
UNTIL <condition>

WHILE <condition> DO
    // statements
ENDWHILE


                
Operators

                
Arithmetic: + - * / MOD DIV

                Relational: = < > <= >= <>

                Logical: AND OR NOT


                
Built-in Functions (Strings & Math)

                
LENGTH(str), LCASE(str), UCASE(str)
SUBSTRING(str, start, length)
DIV(a, b), MOD(a, b)

                        1. Simple INPUT / OUTPUT (REAL data type)
                        
DECLARE MyNumber : REAL
MyNumber <- 3.14
OUTPUT "The value of my number is ", MyNumber

                    


                    

                        2. SELECTION (IF / ELSE)
                        
DECLARE Age : INTEGER
OUTPUT "Enter your age:"
INPUT Age

IF Age >= 18 THEN
    OUTPUT "You are an adult."
ELSE
    OUTPUT "You are a minor."
ENDIF

                    


                    

                        3. ITERATION (FOR, WHILE, REPEAT)
                        
// FOR Loop
DECLARE Counter : INTEGER
FOR Counter <- 1 TO 3
    OUTPUT "Count is: ", Counter
NEXT Counter

// WHILE Loop
DECLARE Number : INTEGER
Number <- 3
WHILE Number > 0 DO
    OUTPUT "Countdown: ", Number
    Number <- Number - 1
ENDWHILE

// REPEAT Loop
DECLARE Password : STRING
REPEAT
    OUTPUT "Enter password:"
    INPUT Password
UNTIL Password = "Secret"
OUTPUT "Access granted!"

                    


                    

                        4. ARRAYS (1D & 2D)
                        
// 1D Array
DECLARE Scores : ARRAY[1:3] OF INTEGER
Scores[1] <- 100
OUTPUT "First score is: ", Scores[1]

// 2D Array (3x3 Grid)
DECLARE Board : ARRAY[1:3, 1:3] OF CHAR
DECLARE Row : INTEGER
DECLARE Col : INTEGER

FOR Row <- 1 TO 3
    FOR Col <- 1 TO 3
        Board[Row, Col] <- '.'
    NEXT Col
NEXT Row

Board[2, 2] <- 'X'
OUTPUT "Center of grid is: ", Board[2, 2]

                    


                    

                        5. FILE HANDLING (Write & Read)
                        
DECLARE WriteText : STRING
DECLARE ReadText : STRING

// Write to virtual file
OPENFILE "MyFile.txt" FOR WRITE
WriteText <- "Hello, virtual file!"
WRITEFILE "MyFile.txt", WriteText
CLOSEFILE "MyFile.txt"

// Read from virtual file
OPENFILE "MyFile.txt" FOR READ
READFILE "MyFile.txt", ReadText
CLOSEFILE "MyFile.txt"

OUTPUT "Data retrieved: ", ReadText

                    


                    

                        6. LIBRARY ROUTINES (RANDOM & ROUND)
                        
DECLARE RandomReal : REAL
DECLARE DiceRoll : INTEGER

// Generate random real number between 0 and 1
RandomReal <- RANDOM()
OUTPUT "Raw Random Number: ", RandomReal

// Generate a random integer between 1 and 6 (Dice Roll)
// 1. RANDOM() * 5 generates a range from 0 to 5.999...
// 2. Adding 1 shifts the range to 1 to 6.999...
// 3. ROUND(..., 0) removes decimals to leave 1, 2, 3, 4, 5, or 6
DiceRoll <- ROUND((RANDOM() * 5) + 1, 0)

OUTPUT "You rolled a ", DiceRoll

                    


                    

                        7. STRING & MATH FUNCTIONS (LCASE, UCASE, SUBSTRING, MOD, DIV)
                        
DECLARE Phrase : STRING
Phrase <- "Computer Science"

// String Functions
OUTPUT "Uppercase: ", UCASE(Phrase)
OUTPUT "Length: ", LENGTH(Phrase)
OUTPUT "First Word: ", SUBSTRING(Phrase, 1, 8)

// Math Functions (DIV and MOD)
// 10 DIV 3 gives 3, 10 MOD 3 gives 1
OUTPUT "10 divided by 3 is ", DIV(10, 3), " remainder ", MOD(10, 3)
